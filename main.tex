\begin{frame}{Motivation}
\framesubtitle{Household travel surveys (HTS) are getting harder to collect}
\begin{columns}
\begin{column}{0.55\textwidth}
\begin{itemize}
\item HTS underpin travel demand models and planning decisions
\item Sample rates: 1--3\% early on, $<$0.5\% by 2006, 0.04\% (CMAP 2019)
\item Low-income, transit-dependent, and minority groups undersampled
\item Self-selection and non-response bias
\end{itemize}
\end{column}

\begin{frame}{Why Agentic AI?}
\begin{itemize}
\item LLMs in transportation are mostly static pipelines (forecasting, imputation, classification)
\begin{itemize}
\item Tasks classical ML already handles
\end{itemize}
\item Agentic AI: LLMs as autonomous agents that reason, remember, and act
\item AL-STARS frames agents as \textbf{simulated travelers}
\begin{itemize}
\item Synthetic, one-to-one, realistic survey responses
\item Context-aware, explainable, behaviorally grounded
\end{itemize}
\item Goal: \emph{complement}, not replace, HTS collection
\end{itemize}
\end{frame}

\begin{frame}{Agent Profile}
\framesubtitle{AL-STARS design}
\begin{enumerate}
\item \textbf{Demographic:} statewide, adds household variables
\item \textbf{Spatial:} tract, urban/rural, transit access
\item \textbf{Attitudes and personality:} sampled conditional on demographics
\item \textbf{Behavioral backstory:} optional
\end{enumerate}
\vspace{1ex}
\textbf{Open questions:} leakage, individual vs.\ household, structured vs.\ narrative format
\end{frame}


\begin{frame}{Memory}
\begin{columns}[t]
\begin{column}{0.33\textwidth}

\begin{itemize}
\item Short-term context + long-term retrieval (RAG)
\item Reflection (Generative Agents, Reflexion)
\end{itemize}
\end{column}
\begin{column}{0.33\textwidth}

\begin{itemize}

\end{itemize}
\end{column}
\begin{column}{0.33\textwidth}
\textbf{AL-STARS}
\begin{itemize}
\item Structured memory of stated facts
\item Long-term memory for multi-day travel diaries
\end{itemize}
\end{column}
\end{columns}
\end{frame}
